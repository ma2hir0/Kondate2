    <script>
        // 初期データ（テスト用）
        let logs = JSON.parse(localStorage.getItem('dinner_logs')) || [
            { id: 1, date: '2026-10-01', menu: 'ハンバーグ', category: '洋食', type: '肉', memo: 'コンソメスープ' },
            { id: 2, date: '2026-10-02', menu: '鯖の塩焼き', category: '和食', type: '魚', memo: '豚汁、豆腐' }
        ];

        // 日付の初期値を今日にする
        document.getElementById('input-date').valueAsDate = new Date();

        function saveLogs() {
            localStorage.setItem('dinner_logs', JSON.stringify(logs));
            renderList();
            updateChart();
        }

        // タブ切り替え
        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(`tab-${tabName}`).classList.remove('hidden');

            document.querySelectorAll('nav button').forEach(el => {
                el.classList.remove('text-indigo-600', 'font-bold');
                el.classList.add('text-gray-500');
            });
            document.getElementById(`nav-${tabName}`).classList.add('text-indigo-600', 'font-bold');
        }

        // フォーム保存
        document.getElementById('dinner-form').addEventListener('submit', (e) => {
            e.preventDefault();
            const newLog = {
                id: Date.now(),
                date: document.getElementById('input-date').value,
                menu: document.getElementById('input-menu').value,
                category: document.getElementById('input-category').value,
                type: document.getElementById('input-type').value,
                memo: document.getElementById('input-memo').value
            };
            logs.unshift(newLog);
            saveLogs();
            document.getElementById('input-menu').value = '';
            document.getElementById('input-memo').value = '';
            alert('保存したよ！');
        });

        // ログ一覧描画
        function renderList() {
            const listEl = document.getElementById('recent-list');
            listEl.innerHTML = '';
            logs.forEach(item => {
                const card = document.createElement('div');
                card.className = 'bg-white p-4 rounded-xl shadow-sm border border-gray-100 flex justify-between items-center';
                card.innerHTML = `
                    <div>
                        <div class="text-xs text-gray-400 mb-0.5">${item.date}</div>
                        <div class="font-bold text-gray-800 text-sm">${item.menu}</div>
                        <div class="text-xs text-gray-500 mt-1">${item.memo || ''}</div>
                    </div>
                    <div class="flex gap-1">
                        <span class="text-xs bg-indigo-50 text-indigo-600 px-2 py-0.5 rounded">${item.category}</span>
                        <span class="text-xs bg-orange-50 text-orange-600 px-2 py-0.5 rounded">${item.type}</span>
                    </div>
                `;
                listEl.appendChild(card);
            });
        }

        // ガチャ機能
        document.getElementById('btn-gacha').addEventListener('click', () => {
            if (logs.length === 0) {
                alert('まだデータがないよ！');
                return;
            }
            const randomItem = logs[Math.floor(Math.random() * logs.length)];
            document.getElementById('gacha-title').textContent = randomItem.menu;
            document.getElementById('gacha-sub').textContent = `${randomItem.category} / ${randomItem.type} (${randomItem.memo || 'メモなし'})`;
            document.getElementById('gacha-result').classList.remove('hidden');
        });

        // グラフ描画
        let chart;
        function updateChart() {
            const counts = { 肉: 0, 魚: 0, その他: 0 };
            logs.forEach(item => counts[item.type] = (counts[item.type] || 0) + 1);

            const ctx = document.getElementById('chart-main').getContext('2d');
            if (chart) chart.destroy();

            chart = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['肉', '魚', 'その他'],
                    datasets: [{
                        data: [counts['肉'], counts['魚'], counts['その他']],
                        backgroundColor: ['#F87171', '#60A5FA', '#9CA3AF']
                    }]
                },
                options: { plugins: { legend: { position: 'bottom' } } }
            });
        }

        // 初期表示
        renderList();
        updateChart();
    </script>
